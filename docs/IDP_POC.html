"""
Step 1-2: Docling Model Download + Document Extraction
Mirrors: genai-idp-extractor → DoclingStrategyProcessor
"""
import io
from dataclasses import dataclass, field
from docling.document_converter import DocumentConverter, PdfFormatOption
from docling.datamodel.base_models import InputFormat
from docling.datamodel.pipeline_options import PdfPipelineOptions, TableFormerMode

# ── Data models (mirrors extraction_models.py in libs/pipeline-events) ──────
@dataclass
class ExtractedPage:
    page_number: int
    text: str                 # full page text (→ page_text.txt in real project)
    word_count: int
    figure_count: int
    table_count: int

@dataclass
class ExtractedDocument:
    source_path: str
    file_type: str
    total_pages: int
    pages: list[ExtractedPage] = field(default_factory=list)
    figures: list[dict]         = field(default_factory=list)
    tables:  list[dict]         = field(default_factory=list)


def build_docling_converter() -> DocumentConverter:
    """
    Mirrors DoclingStrategyProcessor.create_instance().
    Enables: DoclingParseV2 (layout) + RapidOCR (text) + TATR (tables).
    Models auto-download on first call to ~/.cache/docling/
    """
    options = PdfPipelineOptions()
    options.do_ocr = True                              # RapidOCR
    options.do_table_structure = True
    options.table_structure_options.mode = TableFormerMode.ACCURATE  # TATR
    options.images_scale = 2.0                         # higher res for figures
    options.generate_picture_images = True

    print("[Docling] Building converter (models download on first run ~750MB)...")
    return DocumentConverter(
        format_options={InputFormat.PDF: PdfFormatOption(pipeline_options=options)}
    )


# Figure filtering thresholds — mirrors settings.py in the real project
FIGURE_MIN_WIDTH   = 200    # pixels
FIGURE_MIN_HEIGHT  = 50     # pixels
FIGURE_MIN_BYTES   = 16384  # 16 KB
FIGURE_MIN_ASPECT  = 0.1
FIGURE_MAX_ASPECT  = 5.0

def _figure_passes_filter(img_bytes: bytes) -> bool:
    """Post-extraction figure filter (mirrors real project)."""
    if len(img_bytes) < FIGURE_MIN_BYTES:
        return False
    from PIL import Image
    import io
    img = Image.open(io.BytesIO(img_bytes))
    w, h = img.size
    aspect = w / h if h > 0 else 0
    return (w >= FIGURE_MIN_WIDTH and
            h >= FIGURE_MIN_HEIGHT and
            FIGURE_MIN_ASPECT <= aspect <= FIGURE_MAX_ASPECT)


def extract_document(converter: DocumentConverter, pdf_path: str) -> ExtractedDocument:
    """
    Mirrors DoclingStrategyProcessor._process_documents_batch() + _handle_docling_results().
    Returns per-page text, tables (markdown + CSV), and figure info.
    """
    print(f"[Docling] Converting: {pdf_path}")
    result = converter.convert(pdf_path)
    doc    = result.document

    extracted = ExtractedDocument(
        source_path=pdf_path,
        file_type="PDF",
        total_pages=len(doc.pages),
    )
    print(f"[Docling] {len(doc.pages)} pages | {len(doc.tables)} tables | {len(doc.pictures)} pictures")

    # ── Per-page text ─────────────────────────────────────────────────────────
    page_texts: dict[int, list[str]] = {}
    for element, level in doc.iterate_items():
        page_no = element.prov[0].page_no if element.prov else 1
        if hasattr(element, 'text') and element.text:
            page_texts.setdefault(page_no, []).append(element.text)

    # ── Tables ────────────────────────────────────────────────────────────────
    for tbl in doc.tables:
        page_no = tbl.prov[0].page_no if tbl.prov else 1
        try:
            md  = tbl.export_to_markdown()
            csv = tbl.export_to_dataframe().to_csv(index=False)
        except Exception:
            md = csv = ""
        extracted.tables.append({"page": page_no, "markdown": md, "csv": csv})

    # ── Figures ───────────────────────────────────────────────────────────────
    for pic in doc.pictures:
        page_no = pic.prov[0].page_no if pic.prov else 1
        img_bytes = None
        try:
            img = pic.get_image(result)
            if img:
                buf = io.BytesIO(); img.save(buf, format="PNG")
                img_bytes = buf.getvalue()
        except Exception:
            pass
        # Apply post-extraction size filter (mirrors real project)
        if img_bytes and not _figure_passes_filter(img_bytes):
            continue   # decorative / too small → discard
        extracted.figures.append({
            "page": page_no, "image_bytes": img_bytes,
            "caption": str(getattr(pic, 'caption', '')),
            "fig_id":  str(pic.self_ref),
        })

    # ── Build per-page objects ─────────────────────────────────────────────────
    for page in doc.pages:
        pn   = page.page_no
        text = "\n\n".join(page_texts.get(pn, []))
        extracted.pages.append(ExtractedPage(
            page_number=pn,
            text=text,
            word_count=len(text.split()),
            figure_count=sum(1 for f in extracted.figures if f["page"] == pn),
            table_count=sum(1 for t in extracted.tables  if t["page"] == pn),
        ))

    for p in extracted.pages:
        print(f"  page {p.page_number}: {p.word_count} words, {p.figure_count} figs, {p.table_count} tables")
    return extracted
✂️
The real project delegates chunking to AWS Bedrock (FIXED_SIZE, 300 tokens, 20% overlap). The Transformer service only copies .txt/.csv files and writes .metadata.json companions. This POC implements the same chunking in Python so you can see and control the exact behaviour.
"""
Step 3: Fixed-size chunking with overlap
Mirrors: Bedrock KB FIXED_SIZE chunking config (KB_MAX_TOKENS=300, KB_OVERLAP_PERCENTAGE=20%)
Real config in: libs/knowledge-base/src/idp_kb/config.py
"""
from dataclasses import dataclass, field
from typing     import Optional

# Mirrors Bedrock KB chunking config ─────────────────────────────────────────
MAX_TOKENS      = 300    # KB_MAX_TOKENS  — Bedrock config
OVERLAP_PCT     = 0.20   # KB_OVERLAP_PERCENTAGE — 20%
WORDS_PER_TOKEN = 0.75   # rough: 1 token ≈ 0.75 English words


@dataclass
class TextChunk:
    chunk_id:    str
    project_id:  str
    upload_id:   str
    source_path: str
    page_number: int
    source_type: str          # "text" | "table" | "figure_summary"
    text:        str
    metadata:    dict         = field(default_factory=dict)

    # mirrors .metadata.json companion format written by Transformer service
    document_type:   Optional[str] = None  # set after LLM classification
    document_domain: Optional[str] = None
    image_summary:   bool           = False  # True for figure caption chunks


def chunk_text(
    text:        str,
    max_tokens:  int   = MAX_TOKENS,
    overlap_pct: float = OVERLAP_PCT,
) -> list[str]:
    """
    Fixed-size sliding-window chunker.
    Bedrock KB does exactly this server-side during ingestion job.
    We do it here in Python so we can inspect / control each chunk.
    """
    words = text.split()
    if not words:
        return []

    max_words     = max(1, int(max_tokens * WORDS_PER_TOKEN))
    overlap_words = max(1, int(max_words * overlap_pct))
    chunks: list[str] = []
    start = 0

    while start < len(words):
        end   = min(start + max_words, len(words))
        chunks.append(" ".join(words[start:end]))
        if end >= len(words):
            break
        start = end - overlap_words   # slide back by overlap

    return chunks


def build_chunks(
    doc:        ExtractedDocument,
    project_id: str,
    upload_id:  str,
) -> list[TextChunk]:
    """
    Mirrors: Transformer service writes text/table/figure .txt files +
             Bedrock KB ingestion job chunks them during start_ingestion_job().
    We combine both steps here.

    What gets chunked (mirrors Transformer TEXT_FILE_EXTENSIONS = {.txt, .md, .csv}):
      - Page text   → chunk into MAX_TOKENS pieces (source_type="text")
      - Table CSV   → chunk (source_type="table")
      - Figure caption → single chunk each (source_type="figure_summary")
    """
    chunks: list[TextChunk] = []

    # 1. Page text ── mirrors page_text.txt files ─────────────────────────────
    for page in doc.pages:
        if not page.text.strip():
            continue
        for i, chunk_str in enumerate(chunk_text(page.text)):
            chunks.append(TextChunk(
                chunk_id    = f"{upload_id}_p{page.page_number}_txt_{i}",
                project_id  = project_id,
                upload_id   = upload_id,
                source_path = doc.source_path,
                page_number = page.page_number,
                source_type = "text",
                text        = chunk_str,
                metadata    = {"total_pages": doc.total_pages,
                               "chunk_index": i,
                               "file_type": doc.file_type},
            ))

    # 2. Table content ── mirrors table_N.csv / table_N.md files ─────────────
    for j, tbl in enumerate(doc.tables):
        content = tbl.get("markdown", "") or tbl.get("csv", "")
        if not content.strip():
            continue
        for i, chunk_str in enumerate(chunk_text(content)):
            chunks.append(TextChunk(
                chunk_id    = f"{upload_id}_p{tbl['page']}_tbl{j}_{i}",
                project_id  = project_id,
                upload_id   = upload_id,
                source_path = doc.source_path,
                page_number = tbl["page"],
                source_type = "table",
                text        = chunk_str,
                metadata    = {"chunk_index": i},
            ))

    # 3. Figure captions ── mirrors figure_N.txt LLM summary files ───────────
    for fig in doc.figures:
        caption = (fig.get("caption") or "").strip()
        if not caption:
            continue
        chunks.append(TextChunk(
            chunk_id     = f"{upload_id}_p{fig['page']}_fig_{fig['fig_id']}",
            project_id   = project_id,
            upload_id    = upload_id,
            source_path  = doc.source_path,
            page_number  = fig["page"],
            source_type  = "figure_summary",
            text         = caption,
            image_summary= True,
            metadata     = {"fig_id": fig["fig_id"]},
        ))

    txt_ct  = sum(1 for c in chunks if c.source_type == "text")
    tbl_ct  = sum(1 for c in chunks if c.source_type == "table")
    fig_ct  = sum(1 for c in chunks if c.source_type == "figure_summary")
    print(f"[Chunker] {len(chunks)} total  ({txt_ct} text, {tbl_ct} table, {fig_ct} figure)")
    return chunks
🧠
The real project uses amazon.titan-embed-text-v2:0 (512-dim) via AWS Bedrock. The POC offers two options: local sentence-transformers (384-dim, no AWS needed) or the real Bedrock Titan model.
"""
Step 4: Generate embeddings for each chunk.
Real project: Bedrock ingestion job calls titan-embed-text-v2:0 automatically.
POC Option A: local sentence-transformers (no AWS needed)
POC Option B: AWS Bedrock Titan (same as real project)
"""
from typing import Literal

EmbeddingBackend = Literal["local", "bedrock"]

# ── Option A: Local sentence-transformers (no AWS) ────────────────────────────
def embed_local(
    texts:      list[str],
    model_name: str = "all-MiniLM-L6-v2",  # 384-dim
    batch_size: int = 32,
) -> list[list[float]]:
    """Downloads model ~80MB on first call. Fast on CPU."""
    from sentence_transformers import SentenceTransformer
    model = SentenceTransformer(model_name)
    embeddings = model.encode(
        texts,
        show_progress_bar=True,
        batch_size=batch_size,
        normalize_embeddings=True,  # L2 normalize (matches OpenSearch l2 space_type)
    )
    return embeddings.tolist()


# ── Option B: AWS Bedrock Titan (same model as genai-idp) ─────────────────────
def embed_bedrock(
    texts:  list[str],
    region: str = "us-east-1",
    dim:    int = 512,   # KB_EMBEDDING_DIMENSION in the real project
) -> list[list[float]]:
    """
    Model: amazon.titan-embed-text-v2:0
    Same model + dimension used by Bedrock KB ingestion in genai-idp.
    Requires AWS credentials (boto3 default chain).
    """
    import json, boto3
    client = boto3.client("bedrock-runtime", region_name=region)
    embeddings = []
    for text in texts:
        body = json.dumps({
            "inputText":  text[:8192],  # max 8192 tokens for Titan v2
            "dimensions": dim,
            "normalize":  True,
        })
        resp = client.invoke_model(
            modelId="amazon.titan-embed-text-v2:0",
            body=body, contentType="application/json", accept="application/json",
        )
        embeddings.append(json.loads(resp["body"].read())["embedding"])
        print(f"  embedded {len(embeddings)}/{len(texts)}", end="\r")
    print()
    return embeddings


# ── Dispatcher ────────────────────────────────────────────────────────────────
def generate_embeddings(
    texts:   list[str],
    backend: EmbeddingBackend = "local",
) -> tuple[list[list[float]], int]:
    """Returns (embeddings, dimension)."""
    print(f"[Embed] Generating {len(texts)} embeddings via '{backend}'...")
    if backend == "bedrock":
        embeddings = embed_bedrock(texts)
        return embeddings, 512
    else:
        embeddings = embed_local(texts)
        return embeddings, len(embeddings[0]) if embeddings else 384

"""
Step 5-6: Create OpenSearch index + bulk-index chunks with embeddings.
Mirrors: libs/knowledge-base → OpenSearchRepository (index creation)
         Bedrock KB ingestion job (actual chunk indexing)
"""
from opensearchpy            import OpenSearch, RequestsHttpConnection
from opensearchpy.helpers   import bulk

def get_opensearch_client(host: str = "localhost", port: int = 9200) -> OpenSearch:
    return OpenSearch(
        hosts=[{"host": host, "port": port}],
        http_compress=True,
        use_ssl=False,    # True for AWS OpenSearch Serverless (AOSS)
        verify_certs=False,
        connection_class=RequestsHttpConnection,
    )


def create_index_if_not_exists(
    client:     OpenSearch,
    index_name: str,
    dim:        int = 384,
):
    """
    Mirrors OpenSearchRepository.create_index_if_not_exists().
    HNSW/Faiss, L2 distance — exact same settings as the real project.
    The real project names the index by str(project_id) UUID.
    """
    if client.indices.exists(index_name):
        print(f"[OpenSearch] Index '{index_name}' exists — skipping (SYNC mode)")
        return

    body = {
        "settings": {"index": {"knn": True}},
        "mappings": {
            "properties": {
                "vector_field": {          # field name from real project
                    "type":       "knn_vector",
                    "dimension":  dim,
                    "space_type": "l2",      # matches real project
                    "method": {
                        "name":   "hnsw",
                        "engine": "faiss",    # matches real project
                        "parameters": {
                            "ef_construction": 128,  # matches real project
                            "m":              24,   # matches real project
                        },
                    },
                },
                # Metadata fields — mirrors FileMetadata .metadata.json attributes
                "text":        {"type": "text"},
                "chunk_id":    {"type": "keyword"},
                "project_id":  {"type": "keyword"},
                "upload_id":   {"type": "keyword"},
                "page_number": {"type": "integer"},
                "source_type": {"type": "keyword"},  # text|table|figure_summary
                "source_path": {"type": "keyword"},
                "image_summary": {"type": "boolean"},
                "metadata":    {"type": "object"},
            }
        },
    }
    client.indices.create(index_name, body=body)
    print(f"[OpenSearch] Created index '{index_name}' ({dim}-dim HNSW/Faiss/L2)")


def index_chunks(
    client:     OpenSearch,
    index_name: str,
    chunks:     list[TextChunk],
    embeddings: list[list[float]],
    batch_size: int = 100,
):
    """
    Bulk-index chunks.  Each doc mirrors what Bedrock KB ingestion writes to OpenSearch.
    Uses chunk_id as _id → re-indexing the same upload updates docs in place.
    """
    actions = [
        {
            "_index": index_name,
            "_id":    c.chunk_id,
            "_source": {
                "vector_field":  emb,
                "text":          c.text,
                "chunk_id":      c.chunk_id,
                "project_id":   c.project_id,
                "upload_id":    c.upload_id,
                "page_number":  c.page_number,
                "source_type":  c.source_type,
                "source_path":  c.source_path,
                "image_summary":c.image_summary,
                "metadata":     c.metadata,
            },
        }
        for c, emb in zip(chunks, embeddings)
    ]
    total_ok = 0
    for i in range(0, len(actions), batch_size):
        ok, _ = bulk(client, actions[i:i+batch_size])
        total_ok += ok
        print(f"  Batch {i//batch_size+1}: {ok} docs indexed")

    client.indices.refresh(index_name)
    print(f"[OpenSearch] ✓ {total_ok} chunks in '{index_name}'")

def run_pipeline(
    pdf_path:   str,
    project_id: str  = "poc-project-001",
    backend:    str  = "local",
    os_host:    str  = "localhost",
    os_port:    int  = 9200,
):
    upload_id  = str(uuid.uuid4())[:8]
    index_name = project_id.replace("-","_").lower()

    print(f"\n{'='*60}\nIDP POC PIPELINE\nInput:   {pdf_path}\nProject: {project_id}\nUpload:  {upload_id}\nIndex:   {index_name}\n{'='*60}\n")

    # ── Stage 1: Extract ───────────────────────────────────────────────────────
    converter = build_docling_converter()
    doc       = extract_document(converter, pdf_path)

    # ── Stage 2: Chunk ─────────────────────────────────────────────────────────
    chunks = build_chunks(doc, project_id, upload_id)
    if not chunks:
        print("No chunks produced — check the PDF has extractable text.")
        return

    # ── Stage 3: Embed ─────────────────────────────────────────────────────────
    texts      = [c.text for c in chunks]
    embeddings, dim = generate_embeddings(texts, backend=backend)

    # ── Stage 4: Index ─────────────────────────────────────────────────────────
    client = get_opensearch_client(os_host, os_port)
    create_index_if_not_exists(client, index_name, dim=dim)
    index_chunks(client, index_name, chunks, embeddings)

    # ── Stage 5: Search demo ───────────────────────────────────────────────────
    demo_query = "What are the main findings of this document?"
    print(f"\n[Search] Demo query: '{demo_query}'")

    if backend == "bedrock":
        q_emb = embed_bedrock([demo_query])[0]
    else:
        from sentence_transformers import SentenceTransformer
        model = SentenceTransformer("all-MiniLM-L6-v2")
        q_emb = model.encode([demo_query], normalize_embeddings=True)[0].tolist()

    results = hybrid_search(client, index_name, demo_query, q_emb,
                            project_id=project_id, top_k=5)
    for i, r in enumerate(results, 1):
        print(f"  {i}. [p{r['page_number']} | {r['source_type']} | score={r['score']:.3f}]")
        print(f"     {r['text'][:140]}...")

    print(f"\n✅ Done! {len(chunks)} chunks indexed in '{index_name}'")
    print(f"   Check: curl http://localhost:9200/{index_name}/_count")
    return {"chunks": len(chunks), "index": index_name, "dim": dim}

Key point: The OpenSearch index is never recreated on sync. Bedrock's ingestion job calls index_document() for each new chunk — it adds to the existing index. If you re-upload the same file with the same chunk IDs it is an update (upsert by _id). If you upload a new file, the new chunks are appended.

How the RAG app retrieves context
# RAG application pseudocode

def answer_question(project_id, question):
  # 1. Look up KB ID — same table kb-sync writes to
  kb = db.query("""
    SELECT bedrock_kb_id, status
    FROM knowledge_base
    WHERE project_id = %s AND status = 'ACTIVE'
  """, [project_id])

  # 2. Retrieve context from Bedrock KB (hybrid search)
  ctx = bedrock_runtime.retrieve(
    knowledgeBaseId=kb.bedrock_kb_id,
    retrievalQuery={"text": question},
    retrievalConfiguration={
      "vectorSearchConfiguration": {
        "numberOfResults": 5,
        "overrideSearchType": "HYBRID",  # ← hybrid!
      }
    }
  )

  # 3. Build prompt with retrieved chunks
  context = "\n\n".join([r["content"]["text"]
                          for r in ctx["retrievalResults"]])
  return llm.generate(f"Context:\n{context}\n\nQuestion: {question}")
Why no explicit notification is needed
1
Same KB ID, always
The bedrock_kb_id never changes for a project. Once created, all uploads go into the same Bedrock KB. RAG always uses the same ID.
2
Eventual consistency
After start_ingestion_job completes (status=COMPLETE), new chunks are immediately searchable. Bedrock retrieval automatically returns them on the next query.
3
Per-upload status in PostgreSQL
The file_uploads table tracks each upload's status (INDEXED). The API can show "this file is now searchable" but the RAG doesn't check this — it just queries the KB.
4
No index number to update
There is no "index version number". The OpenSearch index is a live mutable store — new chunks appear in search results immediately after ingestion.
🔑
The database mapping is the key: project_id → bedrock_kb_id in PostgreSQL. The RAG application has always known this bedrock_kb_id since KB creation. When kb-sync adds new chunks, they automatically appear in the next retrieval call — no update to the mapping needed.
Q3
What is AWS KB vs OpenSearch — are they the same? + sample code for create index and hybrid search
▾
They are NOT the same — AWS Bedrock Knowledge Base is a managed pipeline layer ON TOP of OpenSearch Serverless (AOSS)
OpenSearch / AOSS
A vector database — stores embeddings + text
You manage: index creation, mapping, embedding, indexing, querying
Supports kNN (vector), BM25 (keyword), hybrid, filters
You call it via REST API / opensearch-py client
Like MySQL — you control the schema and queries
AOSS = serverless version (no node management)
AWS Bedrock Knowledge Base
A managed pipeline — S3 → chunk → embed → index
You just point it at S3 + trigger ingestion; Bedrock handles the rest
Uses OpenSearch Serverless as its backing vector store
You query via bedrock-agent-runtime.retrieve()
Like an ORM (SQLAlchemy) on top of OpenSearch
Also supports Pinecone, Redis as alternatives to AOSS
🔭
Analogy: OpenSearch = MySQL database. AWS Bedrock KB = an ORM framework (like SQLAlchemy) that manages schema creation, migrations, and provides a higher-level query interface. You could bypass the ORM and talk to MySQL directly — that's exactly what this POC does (bypasses Bedrock KB and talks to OpenSearch directly).

# ── Hybrid search: BM25 (keyword) + kNN (vector) combined ─────────────────────
def hybrid_search(
    client,
    index_name:      str,
    query_text:      str,
    query_embedding: list[float],
    project_id:      str       = None,
    top_k:           int       = 5,
    vector_boost:    float     = 1.0,   # weight for vector results
    keyword_boost:   float     = 0.3,   # weight for BM25 results
):
    filters = ["term", {"project_id": project_id}] if project_id else []
    response = client.search(index=index_name, body={
        "size": top_k,
        "query": {
            "bool": {
                "should": [
                    # kNN: find semantically similar chunks
                    {"knn": {"vector_field": {
                        "vector": query_embedding,
                        "k": top_k,
                        "boost": vector_boost,
                    }}},
                    # BM25: find chunks with exact/near keyword matches
                    {"match": {"text": {
                        "query": query_text,
                        "boost": keyword_boost,
                    }}},
                ],
                "minimum_should_match": 1,
                "filter": [{"term": {"project_id": project_id}}] if project_id else [],
            }
        },
        "_source": {"excludes": ["vector_field"]},  # don't return the big vector
    })
    return [{
        "id":          hit["_id"],
        "score":       hit["_score"],
        "text":        hit["_source"]["text"],
        "page_number": hit["_source"]["page_number"],
        "source_type": hit["_source"]["source_type"],
    } for hit in response["hits"]["hits"]]

# ── Example queries ───────────────────────────────────────────────────────────

# 2. Keyword only (BM25)

        "filter": [{"term": {"source_type": "table"}},   # only table chunks
                   {"range": {"page_number": {"gte":1, "lte":10}}}],
    }},
})

Key takeaways
DoclingModel download
Models auto-download to ~/.cache/docling/ (~750MB) on first run. The real project bakes them into Docker at build time. Enable do_ocr=True + TableFormerMode.ACCURATE to match the project's extraction quality.
ChunkingConfig mirrors Bedrock
Bedrock KB uses FIXED_SIZE, 300 max_tokens, 20% overlap. The Transformer service does NOT chunk — it just copies .txt/.csv files. Bedrock's ingestion job chunks server-side. This POC makes that visible by implementing it in Python.
Indexproject_id = index name
The real project names the OpenSearch index by the project's UUID string. One KB per project. The KB ID is stored in PostgreSQL and looked up by both kb-sync AND the RAG application — no notification flow needed.
